\newcommand{\tasksubtaskintro}[1]{%
  \par\vspace{0.5em}\noindent
  For each of the sub-tasks, add a comment statement at the beginning of the code, using the hash symbol `\#', to indicate the sub-task the program code belongs to, for example:\par\vspace{0.6em}\noindent
  \settowidth{\jupytlabelw}{\ttfamily In [1]:\space}%
  \setlength{\fboxsep}{5.5pt}\fboxrule=0.6pt%
  \begin{minipage}[t]{\jupytlabelw}%
    \vspace*{\dimexpr\fboxsep+\fboxrule\relax}\ttfamily In [1]:%
  \end{minipage}%
  \begin{minipage}[t]{\dimexpr\linewidth-\jupytlabelw\relax}%
    \vspace{0pt}%
    \fcolorbox{black}{jupytegray}{%
      \begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}
        \ttfamily\upshape \#Task #1.1\par
        \ttfamily\upshape Program code
      \end{minipage}%
    }\par\vspace{0.2em}%
    \ttfamily Output:%
  \end{minipage}\par\vspace{0.8em}%
}

% --- Ensure total pages is a multiple of 4 with BLANK PAGE immediately below page number ---
\newcommand{\PadToMultipleOfFour}{%
  \clearpage
  \NoTurnOver
  \loop
    \ifnum\numexpr\value{page}-((\value{page}-1)/4)*4\relax=1
    \else
      \vspace*{-0.6em}%
      {\centering\fontsize{12pt}{14pt}\selectfont\textbf{BLANK PAGE}\par}%
      \clearpage
  \repeat
}
