\newcommand{\tasksubtaskintro}[1]{%
  \par\vspace{0.5em}\noindent
  For each of the sub-tasks, add a comment statement at the beginning of the code, using the hash symbol `\#', to indicate the sub-task the program code belongs to, for example:\par\vspace{0.6em}\noindent
  \settowidth{\jupytlabelw}{\ttfamily In [1]:\space}%
  \begin{tabular}{@{}p{\jupytlabelw}@{}p{\dimexpr\linewidth-\jupytlabelw\relax}@{}}
    \ttfamily In [1]: & \setlength{\fboxsep}{5.5pt}\fboxrule=0.6pt%
    \fcolorbox{black}{jupytegray}{%
      \begin{minipage}{\dimexpr\linewidth-\jupytlabelw-14pt\relax}
        \ttfamily\itshape \#Task #1.1\par
        \ttfamily\itshape Program code
      \end{minipage}%
    } \\
    & \vspace{0.2em}\ttfamily Output:
  \end{tabular}\par\vspace{0.8em}%
}

% --- Ensure total pages is a multiple of 4 with BLANK PAGE ---
\newcommand{\PadToMultipleOfFour}{%
  \clearpage
  \NoTurnOver
  \loop
    \ifnum\numexpr\value{page}-((\value{page}-1)/4)*4\relax=1
    \else
      \par\vspace*{-3.0em}%
      {\centering\fontsize{12pt}{14pt}\selectfont\textbf{BLANK PAGE}\par}%
      \clearpage
  \repeat
}
