% --- Ensure total pages is a multiple of 4 with BLANK PAGE ---
\newcommand{\PadToMultipleOfFour}{%
  \clearpage
  \NoTurnOver
  \loop
    \ifnum\numexpr\value{page}-((\value{page}-1)/4)*4\relax=1
    \else
      \vspace*{\fill}
      \begin{center}
        {\fontsize{13pt}{16pt}\selectfont\textbf{BLANK PAGE}}
      \end{center}
      \vspace*{\fill}
      \clearpage
  \repeat
}
